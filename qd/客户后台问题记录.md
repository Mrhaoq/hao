``` js
async function fetchData(API, queryParams) {
  try {
    this.loading = true;
    const res = await API(queryParams);
    if (res.code != 200) return;
    this.list = res.data;
    this.total = res.total;
  } catch (error) {
    console.log(error);
  } finally {
    this.loading = false;
  }
}
```
