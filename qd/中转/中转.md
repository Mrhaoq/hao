<template>
  <div class="home">
    <li class="home-title">入库吨数</li>
    <div class="area-container">
      <el-form
        size="small"
        :inline="true"
        label-width="68px"
        style="margin-top: 30px"
      >
        <el-form-item label="中心仓名称/实际公司名称" label-width="200">
          <el-input
            v-model="queryParams.recoveryName"
            placeholder="请输入中心仓名称"
            clearable
            @keyup.enter.native="search"
          />
        </el-form-item>
        <el-form-item label="" label-width="200">
          <el-input
            v-model="queryParams.storehouse"
            placeholder="请输入实际公司名称"
            clearable
            @keyup.enter.native="search"
          />
        </el-form-item>

​        <el-button type="primary" class="btn-bc btn-size" @click="search"
​          >搜索</el-button
​        ><el-button plain class="btn-size btn" @click="reset">重置</el-button>
​      </el-form>
​    </div>

    <div class="block mt20">
      <el-table
        :data="tableList"
        table-layout="auto"
        class="tableAuto"
        show-summary
        :summary-method="getSummaries"
        style="border-radius: 6px"
        :header-cell-style="{ background: '#d7e3fd', color: '#638ef8' }"
        v-loading="loading"
      >
        <el-table-column type="index" label="序号" align="center" width="60" />
        <el-table-column
          prop="recoveryName"
          min-width="180"
          align="center"
          label="中心仓名称"
        >
          <template slot-scope="scope">
            {{ scope.row.recoveryName ? scope.row.recoveryName : "-" }}
          </template>
        </el-table-column>
        <el-table-column
          prop="storehouse"
          min-width="180"
          align="center"
          label="实际公司名称"
        >
          <template slot-scope="scope">
            {{ scope.row.storehouse ? scope.row.storehouse : "-" }}
          </template>
        </el-table-column>
        <el-table-column min-width="160" align="center" label="入库类型">
          <template slot-scope="scope">
            <el-switch
              v-model="scope.row.type"
              :active-value="1"
              :inactive-value="2"
              active-text="线上"
              inactive-text="线下"
              @change="handleStatusChange(scope.row)"
            ></el-switch>
          </template>
        </el-table-column>

​        <el-table-column
​          fixed="right"
​          prop="realWeight"
​          min-width="160"
​          align="center"
​          label="本次入库吨数"
​        >
          <template slot-scope="scope">
            <el-input v-model="scope.row.realWeight" />
          </template>
​        </el-table-column>
​      </el-table>
​    </div>

    <div class="block mt20">
      <el-table
        :data="tableList"
        table-layout="auto"
        class="tableAuto"
        show-summary
        :summary-method="getSummaries"
        style="border-radius: 6px"
        :header-cell-style="{ background: '#d7e3fd', color: '#638ef8' }"
        v-loading="loading"
      >
        <el-table-column type="index" label="序号" align="center" width="60" />
        <el-table-column
          prop="recoveryName"
          min-width="180"
          align="center"
          label="中心仓名称"
        >
          <template slot-scope="scope">
            {{ scope.row.recoveryName ? scope.row.recoveryName : "-" }}
          </template>
        </el-table-column>
        <el-table-column
          prop="storehouse"
          min-width="180"
          align="center"
          label="实际公司名称"
        >
          <template slot-scope="scope">
            {{ scope.row.storehouse ? scope.row.storehouse : "-" }}
          </template>
        </el-table-column>
        <el-table-column
          prop="recoveryWeight"
          min-width="120"
          align="center"
          label="待入库重量(吨)"
        >
          <template slot-scope="scope">
            {{
              scope.row.recoveryWeight
                ? (scope.row.recoveryWeight / 1000).toFixed(3)
                : "-"
            }}
          </template>
        </el-table-column>
        <el-table-column
          prop="orderCount"
          min-width="100"
          align="center"
          label="剩余订单数量"
        >
          <template slot-scope="scope">
            {{ scope.row.orderCount ? scope.row.orderCount : "-" }}

        <el-table-column

          <template slot-scope="scope">

        <el-table-column

        >
          <template slot-scope="scope">
            {{
              scope.row.todayTotalWeight
                ? (scope.row.todayTotalWeight / 1000).toFixed(3)
                : 0
            }}
          </template>
        </el-table-column>
        <el-table-column min-width="160" align="center" label="入库类型">
          <template slot-scope="scope">
            <el-switch
              v-model="scope.row.type"
              :active-value="1"
              :inactive-value="2"
              active-text="线上"
              inactive-text="线下"
              @change="handleStatusChange(scope.row)"

        </el-table-column>

​        <el-table-column
​          fixed="right"
​          prop="realWeight"
​          min-width="160"
​          align="center"
​          label="本次入库吨数"
​        >
          <template slot-scope="scope">
            <el-input v-model="scope.row.realWeight" />

    <div class="flex-between mt20">

<script>
import { applyInsertStock } from "@/api/audit/index.js";
import { businessList } from "@/api/environmental";
import { format_kg_num } from "@/utils/validation";
export default {
  name: "Environmental",
  data() {
    return {
      // 列表
      tableList: [],
      // 地区id
      value: "",
      // 地区选项
      options: [],
      // 参数
      queryParams: {
        applyDetailRequestVOS: [],
      },
      // 发送参数
      params: {},

​      loading: true,
​    };
  },
  created() {
​    this.getList();
  },
  watch: {
​    tableList: {
​      handler(newVal) {
​        this.params.applyDetailRequestVOS = newVal
​          .filter((i) => Number(i.realWeight))
​          .map((item) => {
​            item.realWeight = format_kg_num(item.realWeight);
​            return {
​              recoveryId: item.id,
​              weight: item.realWeight,
​              type: item.type,
​              recoveryName: item.recoveryName,
​              storehouse: item.storehouse,
​            };
​          });
​      },
​      deep: true,
​    },
  },
  methods: {
​    // 获取列表
​    getList() {
​      this.loading = true;
​      return businessList(this.queryParams).then((res) => {
​        if (!(res.code == 200)) return;
​        this.tableList = res.rows;
​        this.loading = false;
​      });
​    },
​    reset() {
​      this.queryParams = {};
​      this.search();
​    },
​    // 搜索
​    async search() {
​      await this.getList();
​      this.$message({ message: "搜索成功", type: "success" });
​    },

​    // 提交
​    submit() {
​      if (!this.params.applyDetailRequestVOS.length)
​        return this.$message.error("本次入库总吨数不能为 0");
​      this.$confirm(`此操作将提交本次入库重量，是否确认?`, "系统提示", {
​        confirmButtonText: "确定",
​        cancelButtonText: "取消",
​        closeOnClickModal: false,
​        type: "info",
​      }).then(() => {
​        this.loading = true;
​        applyInsertStock(this.params).then((res) => {
​          this.getList();
​          this.loading = false;
​          this.$message({ message: `保存成功`, type: "success" });
​        });
​      });
​    },
​    // 总计
​    getSummaries(param) {
​      const { columns, data } = param;
​      const sums = [];
​      columns.forEach((column, index) => {
​        if (index === 0) {
​          sums[index] = "总计";
​          return;
​        }
​        const values = data.map((item) => Number(item[column.property]));
​        if (!values.every((value) => isNaN(value))) {
​          sums[index] = values.reduce((prev, curr) => {
​            const value = Number(curr);
​            if (!isNaN(value)) {
​              return prev + curr;
​            } else {
​              return prev;
​            }
​          }, 0);

​          sums[index] =
​            column.property == "orderCount"
​              ? sums[index] + " 单"
​              : column.property == "storehouse"
​              ? (sums[index] = "N/A")
​              : column.property == "realWeight"
​              ? sums[index].toFixed(3) + " 吨"
​              : (sums[index] / 1000).toFixed(3) + " 吨";
​        } else {
​          sums[index] = "N/A";
​        }
​      });

​      return sums;
​    },
  },
};
</script>
