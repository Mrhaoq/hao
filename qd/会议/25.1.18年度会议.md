   <script setup lang="ts">
   import { invoiceUpdate } from "@/api/agent/invoice";
   import { onLoad, onShow } from "@dcloudio/uni-app";
   import { ref } from "vue";
   import { useUserStore } from "@/store/modules/user";
   const userStore:any = useUserStore();

   const radioList = ref([
     {
       name: "invoice",
       label: "开票",
       bool: true,
     },
     {
       name: "unInvoice",
       label: "不开票",
       bool: false,
     },
   ]);
   const invoiceFlag = ref(false);
   const form = ref({
     companyName: "",
     phone: "",
     name: "",
     normalAddress: "",
     idCardNum: "",
     idCardAddress: "",
   });
   const realData = ref({});
   // 切换开票状态
   const invoiceChange = (val: any) => {
     if (val === "unInvoice") {
       my.showModal({
         title: "温馨提示",
         content: "切换后，将无法开票",
         confirmText: "确认切换",
         success: (res: any) => {
           console.log(res);
           if (res.confirm) {
             invoiceFlag.value = false;
             confirm();
           } else {
             invoiceFlag.value = true;
           }
         },
         fail: () => {
           invoiceFlag.value = true;
         },
       });
     } else {
       const real: any = userStore.orgAuthReal;
       if (real && real.status == 1) {
         invoiceFlag.value = true;
         form.value = real;
       } else {
         invoiceFlag.value = false;
         my.showToast({
           content: "请先实名认证",
           icon: "none",
         });
       }
     }
   };
   // 确认发票抬头
   const confirm = async () => {
     const data = {
       invoiceFlag: invoiceFlag.value,
     };
     const res: any = await invoiceUpdate(data);
     if (res.code == 200) userStore.getInvoiceConfig();
     if (invoiceFlag.value) {
       my.redirectTo({
         url: "/pages-agent/pages/invoice/invoice",
       });
     }
   };
   onShow(() => {
     const real: any = userStore.orgAuthReal;
     console.log(real);
     if (real) form.value = real;
   });
   onLoad(() => {
     invoiceFlag.value = userStore.invoiceInfo?.invoiceFlag || false;
   });
   </script>
