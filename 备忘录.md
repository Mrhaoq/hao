<template>
  <view class="page-container">
    <!-- 头部 -->
    <view v-if="showHeader" :style="headerStyle" class="page-header">
      <view class="header-content" :style="headerContentStyle">
        <slot name="fixed-header"></slot>
      </view>
    </view>

​    <!-- 内容区域 -->
​    <view class="page-content" :style="contentStyle">
​      <!-- 内容包装器 -->
​      <view class="content-wrapper">
​        <!-- 主内容区域 -->
​        <view class="main-content">
​          <slot></slot>
​        </view>

​        <!-- 支持文本 -->
​        <view
​          class="support-text-wrapper"
​          :style="{ marginBottom: showFooter ? footerTotalHeight : '20rpx' }"
​        >
​          <text class="support-text">千源百荟 提供技术支持</text>
​        </view>
​      </view>
​    </view>

​    <!-- 底部固定区域 -->
​    <view v-if="showFooter" class="page-footer" :style="footerStyle">
​      <view class="footer-content" :style="footerContentStyle">
​        <slot name="fixed-footer"></slot>
​      </view>
​    </view>
  </view>
</template>

<script setup lang="ts">
import { computed } from "vue";
import type { CSSProperties } from "vue";

interface StyleProps {
  height?: string;
  backgroundColor?: string;
  opacity?: number;
  zIndex?: number;
  showShadow?: boolean;
  padding?: string;
}

interface LayoutBasicProps {
  showHeader?: boolean;
  showFooter?: boolean;
  headerStyle?: StyleProps;
  footerStyle?: StyleProps;
  pagePadding?: string;
  pageBackground?: string;
}

// 获取系统信息
const getSystemInfo = () => {
  try {
    return my.getSystemInfoSync();
  } catch (error) {
    console.error("获取系统信息失败:", error);
    return { statusBarHeight: 0 };
  }
};

// 计算头部总高度
const headerTotalHeight = computed(() => {
  if (!props.showHeader) return "0";
  const baseHeight = props.headerStyle?.height || "88rpx";
  return `calc(${statusBarHeight.value}px + ${baseHeight} + 20rpx)`;
});

const props = withDefaults(defineProps<LayoutBasicProps>(), {
  showHeader: false,
  showFooter: false,
  headerStyle: () => ({
    backgroundColor: "#ffffff",
    opacity: 1,
    zIndex: 100,
    showShadow: true,
    padding: "0",
  }),
  footerStyle: () => ({
    backgroundColor: "#ffffff",
    opacity: 1,
    zIndex: 100,
    showShadow: true,
    padding: "0 30rpx",
  }),
  pagePadding: "30rpx",
  pageBackground: "#f6f7fa",
});

// 状态栏高度
const statusBarHeight = computed(() => {
  const info = getSystemInfo();
  return 0;
});

// 是否需要安全区域
const needSafeArea = computed(() => {
  const systemInfo = getSystemInfo();
  return systemInfo.model?.includes("iPhone") || false;
});

// 计算底部总高度
const footerTotalHeight = computed(() => {
  const baseHeight = props.footerStyle?.height || "88rpx";
  const safeAreaHeight = needSafeArea.value ? "34px" : "0";
  return `calc(${baseHeight} + ${safeAreaHeight})`;
});

// header样式
const headerStyle = computed<CSSProperties>(() => ({
  backgroundColor: props.headerStyle?.backgroundColor,
  zIndex: props.headerStyle?.zIndex,
  boxShadow: props.headerStyle?.showShadow
    ? "0 2rpx 10rpx rgba(0,0,0,0.05)"
    : "none",
  paddingTop: `${statusBarHeight.value}px`,
}));

// header内容样式
const headerContentStyle = computed<CSSProperties>(() => ({
  padding: props.headerStyle?.padding,
  height: props.headerStyle?.height,
  minHeight: "88rpx",
  paddingBottom: "20rpx",
}));

// footer样式
const footerStyle = computed<CSSProperties>(() => ({
  backgroundColor: props.footerStyle?.backgroundColor,
  zIndex: props.footerStyle?.zIndex,
  boxShadow: props.footerStyle?.showShadow
    ? "0 2rpx 10rpx rgba(0,0,0,0.05)"
    : "none",
}));

// footer内容样式
const footerContentStyle = computed<CSSProperties>(() => ({
  padding: props.footerStyle?.padding,
  height: props.footerStyle?.height,
  minHeight: "88rpx",
  paddingTop: "20rpx",
}));

// 内容区样式
const contentStyle = computed<CSSProperties>(() => {
  const headerTotalHeight = props.showHeader
    ? `calc(${statusBarHeight.value}px + ${
        props.headerStyle?.height || "88rpx"
      } + 20rpx)`
    : "0";

  return {
    paddingTop: props.showHeader ? headerTotalHeight : props.pagePadding,
    paddingLeft: props.pagePadding,
    paddingRight: props.pagePadding,
    backgroundColor: props.pageBackground,
  };
});
</script>
