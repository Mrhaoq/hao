const getQrcode = async () => {
  if (!props.record.orderId || !props.showOperation) return;

  // 是否居间单
  const isBrokerOrder = props.record?.comboServiceOrderDTOList?.[0]?.serviceType == 31;
  console.log("isBrokerOrder", isBrokerOrder);
  const miniInfo = await getSaasOrgData();
  /* 
    判断是否居间单，如果是居间单跳转合同二维码url，否则跳转知情同意书二维码url，
    在判断是否居间单的基础上，再判断是否是 saas租户。如果是 saas租户跳转挚护易url，自有租户跳转挚护到家url
    miniInfo.isSaasOrg true: SaaS 家政用户；false: 自有租户
      挚护到家合同、知情同意书
      https://yza.cn/zhdjcontract     https://yza.cn/agreement
      挚护易合同、知情同意书  saas租户跳转
      https://yza.cn/zhycontract    https://yilife.cn/agreement
  */
  let url = "";
  if (isBrokerOrder) {
    url = miniInfo.isSaasOrg ? `https://yza.cn/zhycontract?orderId=${props.record.orderId}` : `https://yza.cn/zhdjcontract?orderId=${props.record.orderId}`;
  } else {
    url = miniInfo.isSaasOrg ? `https://yilife.cn/agreement?orderId=${props.record.orderId}` : `https://yza.cn/agreement?orderId=${props.record.orderId}`;
  }
  console.log("居间单跳转二维码url  >>>   ", url);
  const res = await api3239({
    url,
    userId: props.record?.userId,
    type: 2,
  });
  qrCodeUrl.value = res.data || "";
};
