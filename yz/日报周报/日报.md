const getQrcode = async () => {
  if (!props.record.orderId || !props.showOperation) return;

  // 是否居间单
  const isBrokerOrder = props.record?.comboServiceOrderDTOList?.[0]?.serviceType == 31;
  console.log("isBrokerOrder", isBrokerOrder);
  const miniInfo = await getSaasOrgData();

  /* 
      首先判断是否居间单，其次判断是否是saas租户。saas租户跳转挚护易url，自有租户跳转挚护到家url
        挚护到家合同、知情同意书
        https://yza.cn/agreement
        https://yza.cn/zhdjcontract
        挚护易合同、知情同意书  saas租户跳转
        https://yza.cn/zhycontract
        https://yilife.cn/agreement
  */
  let url = isBrokerOrder ? `https://yza.cn/zhycontract?orderId=${props.record.orderId}` : `https://yza.cn/agreement?orderId=${props.record.orderId}`;
  if (isBrokerOrder) {
    url = `https://yza.cn/zhycontract?orderId=${props.record.orderId}`
    if(miniInfo.isSaasOrg){
      url =  `https://yza.cn/agreement?orderId=${props.record.orderId}`;
    }
  } else {
    url = `https://yza.cn/zhdjcontract?orderId=${props.record.orderId}`;
    if(miniInfo.isSaasOrg){
      url = `https://yilife.cn/agreement?orderId=${props.record.orderId}`;
    }
  }
  const res = await api3239({
    url,
    userId: props.record?.userId,
    type: 2,
  });
  qrCodeUrl.value = res.data || "";
};
