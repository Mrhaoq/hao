权限包含部分就可以

```js
const roleArr = ["000001", "000002", "000003", "000004", "000005"]; // 请求回的已有权限
const role = ["000001", "013"]; // 当前按钮或其他的操作功能所需权限

// 包含其中一个就可以操作
const hasPremi = premiArr => {
	if (typeof premiArr == "string") {
		return roleArr.includes(premiArr);
	} else if (Array.isArray(premiArr)) {
		for (let i = 0; i < premiArr.length; i++) {
			const element = premiArr[i];
			if (roleArr.includes(element)) {
				return true;
			}
		}
	}
	return false;
};

console.log(hasPremi(role));
```
